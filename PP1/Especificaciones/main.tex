\begin{minipage}[t]{0.62\textwidth}
  \textbf{Lista de CIs y componentes principales.}
  \vspace{2pt}
  \footnotesize
  \begin{tabularx}{\linewidth}{lX}
    \toprule
    \textbf{CI / ref.}  & \textbf{Funci\'{o}n y conexi\'{o}n esencial}\\
    \midrule
    74LS283 & Sumador binario; $C_\text{in}\!\to\!\mathrm{GND}$, $\Sigma_0{=}1$, $\Sigma_1{=}4$, $\Sigma_2{=}13$.\\
    74LS86  & XOR: pines 1--3 ($A_0,B_0\!\to\!X_0$); 4--6 ($A_1,B_1\!\to\!X_1$).\\
    74LS08  & AND: pines 1--3 ($A_0,B_0\!\to\!N_0$); 4--6 ($A_1,B_1\!\to\!N_1$).\\
    74LS85  & Comparador; $A$ en 10/12, $B$ en 9/11; cascada a GND/VCC; $A > B$ en pin 5.\\
    74LS157 & MUX 2-a-1; select com\'{u}n a ($A > B \lor A = B$). Salidas en 4 y 7.\\
    74LS04  & Inversor; pin 1 a EN, pin 2 entrega $\overline{\mathrm{EN}}$.\\
    74LS139 & Decoder; $\overline{Y_i}$ en 4--7; $\overline{G}$ en 1.\\
    74LS153 & MUX 4-a-1; entradas 1C0..1C3 en pines 3--6; $1Y$ en 7.\\
    74LS47  & Driver BCD $\to$ 7 segmentos; entradas BCD en 7,1,2,6.\\
    Display & 7-segmentos, \'{a}nodo com\'{u}n. \'{A}nodos (3, 8) a $+5\,\mathrm{V}$.\\
    \bottomrule
  \end{tabularx}
\end{minipage}%
